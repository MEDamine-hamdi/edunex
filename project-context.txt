package com.edunex.backend.entity;

import jakarta.persistence.*;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;

import java.time.LocalDateTime;

@Entity
@Table(name = "activity_logs")
@Getter
@Setter
@NoArgsConstructor
public class ActivityLog {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(nullable = false, length = 200)
    private String action;

    @Column(name = "entity_type", length = 100)
    private String entityType;

    @Column(nullable = false, updatable = false)
    private LocalDateTime timestamp;

    @Column(name = "ip_address", length = 60)
    private String ipAddress;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "user_id")
    private User user;

    @PrePersist
    protected void onCreate() {
        this.timestamp = LocalDateTime.now();
    }
}

package com.edunex.backend.entity;

import com.edunex.backend.enums.TargetAudience;
import jakarta.persistence.*;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;

import java.time.LocalDateTime;

@Entity
@Table(name = "announcements")
@Getter
@Setter
@NoArgsConstructor
public class Announcement {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(nullable = false, length = 200)
    private String title;

    @Column(nullable = false, columnDefinition = "TEXT")
    private String content;

    @Column(name = "published_at", nullable = false, updatable = false)
    private LocalDateTime publishedAt;

package com.edunex.backend.security;

import com.edunex.backend.service.ActivityLogService;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.springframework.security.authentication.AnonymousAuthenticationToken;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.servlet.HandlerInterceptor;

/** Records every successful write request (POST, PUT, PATCH, DELETE) made by an authenticated user. */
@Component
@RequiredArgsConstructor
public class ActivityLogInterceptor implements HandlerInterceptor {

    private final ActivityLogService activityLogService;

    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response,
                                Object handler, Exception ex) {
        String method = request.getMethod();
        if (method.equals("GET") || method.equals("HEAD") || method.equals("OPTIONS")) {
            return;
        }
        if (response.getStatus() >= 400) {
            return;
        }
        Authentication auth = SecurityContextHolder.getContext().getAuthentication();
        if (auth == null || !auth.isAuthenticated() || auth instanceof AnonymousAuthenticationToken) {
            return;
        }
        try {
            String path = request.getRequestURI();
            String verb = switch (method) {
                case "POST" -> "CREATE";
                case "DELETE" -> "DELETE";
                default -> "UPDATE";
            };
            String[] parts = path.split("/");
            String entity = parts.length > 3 ? parts[3] : parts[parts.length - 1];
            activityLogService.registerForEmail(verb + " " + path, entity, auth.getName(), request.getRemoteAddr());
        } catch (Exception ignored) {
            // logging must never break the request
        }
    }
}

===== backend\src\main\java\com\edunex\backend\security\CustomUserDetailsService.java =====
package com.edunex.backend.security;

import com.edunex.backend.entity.User;
import com.edunex.backend.enums.UserStatus;
import com.edunex.backend.repository.UserRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.core.userdetails.UsernameNotFoundException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.HashSet;
import java.util.Set;

@Service
@RequiredArgsConstructor
public class CustomUserDetailsService implements UserDetailsService {

    private final UserRepository userRepository;

    @Override
    @Transactional(readOnly = true)
    public UserDetails loadUserByUsername(String email) throws UsernameNotFoundException {
        User user = userRepository.findByEmail(email)
                .orElseThrow(() -> new UsernameNotFoundException("User not found"));

        Set<GrantedAuthority> authorities = new HashSet<>();
        if (user.getRole() != null) {
            authorities.add(new SimpleGrantedAuthority(user.getRole().getName()));
            user.getRole().getPermissions()
                    .forEach(p -> authorities.add(new SimpleGrantedAuthority(p.getCode())));
        }

        return org.springframework.security.core.userdetails.User.builder()
                .username(user.getEmail())
                .password(user.getPassword())
                .authorities(authorities)
                .disabled(user.getStatus() != UserStatus.ACTIVE)
                .build();
    }
}


===== backend\src\main\java\com\edunex\backend\security\JwtAuthenticationFilter.java =====
package com.edunex.backend.security;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.security.core.userdetails.UsernameNotFoundException;
import org.springframework.security.web.authentication.WebAuthenticationDetailsSource;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;

@Component
@RequiredArgsConstructor
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    private final JwtService jwtService;
    private final CustomUserDetailsService userDetailsService;

    @Override
    protected void doFilterInternal(HttpServletRequest request,
                                    HttpServletResponse response,
                                    FilterChain filterChain) throws ServletException, IOException {

        String header = request.getHeader("Authorization");

        if (header != null && header.startsWith("Bearer ")) {
            String token = header.substring(7);

            if (jwtService.isTokenValid(token)
                    && SecurityContextHolder.getContext().getAuthentication() == null) {
                try {
                    String email = jwtService.extractEmail(token);
                    UserDetails userDetails = userDetailsService.loadUserByUsername(email);

                    if (userDetails.isEnabled()) {
                        UsernamePasswordAuthenticationToken authentication =
                                new UsernamePasswordAuthenticationToken(
                                        userDetails, null, userDetails.getAuthorities());
                        authentication.setDetails(
                                new WebAuthenticationDetailsSource().buildDetails(request));
                        SecurityContextHolder.getContext().setAuthentication(authentication);
                    }
                } catch (UsernameNotFoundException ignored) {
                    // unknown user: request stays unauthenticated
                }
            }
        }

        filterChain.doFilter(request, response);
    }
}


===== backend\src\main\java\com\edunex\backend\security\JwtService.java =====
package com.edunex.backend.security;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.JwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.io.Decoders;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import javax.crypto.SecretKey;
import java.util.Date;

@Service
public class JwtService {

    private final SecretKey key;
    private final long expirationMs;

    public JwtService(@Value("${jwt.secret}") String secret,
                      @Value("${jwt.expiration-ms}") long expirationMs) {
        this.key = Keys.hmacShaKeyFor(Decoders.BASE64.decode(secret));
        this.expirationMs = expirationMs;
    }

    public String generateToken(String email, String role) {
        Date now = new Date();
        return Jwts.builder()
                .subject(email)
                .claim("role", role)
                .issuedAt(now)
                .expiration(new Date(now.getTime() + expirationMs))
                .signWith(key)
                .compact();
    }

    public String extractEmail(String token) {
        return parseClaims(token).getSubject();
    }

    public String extractRole(String token) {
        return parseClaims(token).get("role", String.class);
    }

    public boolean isTokenValid(String token) {
        try {
            parseClaims(token);
            return true;
        } catch (JwtException | IllegalArgumentException e) {
            return false;
        }
    }

    private Claims parseClaims(String token) {
        return Jwts.parser()
                .verifyWith(key)
                .build()
                .parseSignedClaims(token)
                .getPayload();
    }
}


===== backend\src\main\java\com\edunex\backend\security\LoginAttemptService.java =====
package com.edunex.backend.security;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import java.time.Duration;
import java.time.Instant;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

/** In-memory brute-force protection: too many failures for a key block it for a while. */
@Component
public class LoginAttemptService {

    private record Attempt(int failures, Instant lastFailure) {
    }

    private final int maxFailures;
    private final Duration lockDuration;
    private final Map<String, Attempt> attempts = new ConcurrentHashMap<>();

    public LoginAttemptService(@Value("${app.security.login-max-failures:5}") int maxFailures,
                               @Value("${app.security.login-lock-minutes:15}") long lockMinutes) {
        this.maxFailures = maxFailures;
        this.lockDuration = Duration.ofMinutes(lockMinutes);
    }

    public boolean isBlocked(String key) {
        return isBlocked(key, Instant.now());
    }

    boolean isBlocked(String key, Instant now) {
        Attempt a = attempts.get(key);
        if (a == null) {
            return false;
        }
        if (!now.isBefore(a.lastFailure().plus(lockDuration))) {
            attempts.remove(key);
            return false;
        }
        return a.failures() >= maxFailures;
    }

    /** Returns true when this failure triggers (or keeps) the lock. */
    public boolean recordFailure(String key) {
        return recordFailure(key, Instant.now());
    }

    boolean recordFailure(String key, Instant now) {
        if (attempts.size() > 10_000) {
            attempts.entrySet().removeIf(e -> !now.isBefore(e.getValue().lastFailure().plus(lockDuration)));
        }
        Attempt updated = attempts.merge(key, new Attempt(1, now), (old, ignored) ->
                now.isBefore(old.lastFailure().plus(lockDuration))
                        ? new Attempt(old.failures() + 1, now)
                        : new Attempt(1, now));
        return updated.failures() >= maxFailures;
    }

    public long retryAfterSeconds(String key) {
        return retryAfterSeconds(key, Instant.now());
    }

    long retryAfterSeconds(String key, Instant now) {
        Attempt a = attempts.get(key);
        if (a == null) {
            return 0;
        }
        return Math.max(0, Duration.between(now, a.lastFailure().plus(lockDuration)).getSeconds());
    }

    public void reset(String key) {
        attempts.remove(key);
    }
}

package com.edunex.backend.config;

import com.edunex.backend.entity.PlatformAdministrator;
import com.edunex.backend.entity.Role;
import com.edunex.backend.repository.PlatformAdministratorRepository;
import com.edunex.backend.repository.RoleRepository;
import com.edunex.backend.repository.UserRepository;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.CommandLineRunner;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Component;

import java.util.Map;

@Slf4j
@Component
@RequiredArgsConstructor
public class DataInitializer implements CommandLineRunner {

    private final RoleRepository roleRepository;
    private final UserRepository userRepository;
    private final PlatformAdministratorRepository platformAdministratorRepository;
    private final PasswordEncoder passwordEncoder;

    @Value("${app.admin.email:}")
    private String adminEmail;

    @Value("${app.admin.password:}")
    private String adminPassword;

    @Override
    public void run(String... args) {
        Map<String, String> roles = Map.of(
                "ROLE_STUDENT", "University student",
                "ROLE_PROFESSOR", "University professor",
                "ROLE_ADMIN_AGENT", "University administration agent",
                "ROLE_PLATFORM_ADMIN", "EduNex platform administrator"
        );

        roles.forEach((name, description) -> {
            if (roleRepository.findByName(name).isEmpty()) {
                Role role = new Role();
                role.setName(name);
                role.setDescription(description);
                roleRepository.save(role);
                log.info("Role created: {}", name);
            }
        });

        if (adminEmail.isBlank() || adminPassword.isBlank()) {
            log.warn("app.admin.email / app.admin.password not set: no platform admin created");
            return;
        }

        if (userRepository.existsByEmail(adminEmail)) {
            return;
        }

        PlatformAdministrator admin = new PlatformAdministrator();
        admin.setFirstName("Platform");
        admin.setLastName("Admin");
        admin.setEmail(adminEmail);
        admin.setPassword(passwordEncoder.encode(adminPassword));
        admin.setRole(roleRepository.findByName("ROLE_PLATFORM_ADMIN").orElseThrow());
        platformAdministratorRepository.save(admin);
        log.info("First platform administrator created: {}", adminEmail);
    }
}

import java.util.Map;

@Slf4j
@Component
@RequiredArgsConstructor
public class PermissionInitializer implements CommandLineRunner {

    private static final Map<String, String> ROLE_DESCRIPTIONS = Map.of(
            "ROLE_STUDENT", "University student",
            "ROLE_PROFESSOR", "University professor",
            "ROLE_ADMIN_AGENT", "University administration agent",
            "ROLE_PLATFORM_ADMIN", "EduNex platform administrator");

    private final RoleRepository roleRepository;
    private final PermissionRepository permissionRepository;

    @Override
    @Transactional
    public void run(String... args) {
        for (PermissionCatalog.Entry entry : PermissionCatalog.ENTRIES) {
            if (permissionRepository.findByCode(entry.code()).isPresent()) {
                continue;
            }
            Permission permission = new Permission();
            permission.setCode(entry.code());
            permission.setDescription(entry.description());
            permission = permissionRepository.save(permission);

            for (String roleName : entry.roles()) {
                Role role = roleRepository.findByName(roleName).orElseGet(() -> {
                    Role created = new Role();
                    created.setName(roleName);
                    created.setDescription(ROLE_DESCRIPTIONS.get(roleName));
                    return roleRepository.save(created);
                });
                role.getPermissions().add(permission);
                roleRepository.save(role);
            }
            log.info("Permission created: {}", entry.code());
        }
    }
}

import org.springframework.security.web.SecurityFilterChain;

import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;

@Configuration
@EnableWebSecurity
@EnableMethodSecurity
@RequiredArgsConstructor
public class SecurityConfig {

    private final JwtAuthenticationFilter jwtAuthenticationFilter;

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        http
                .csrf(csrf -> csrf.disable())
                .sessionManagement(session ->
                        session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
                .authorizeHttpRequests(auth -> auth

                .exceptionHandling(ex ->

                .addFilterBefore(jwtAuthenticationFilter, UsernamePasswordAuthenticationFilter.class);

        return http.build();

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }

    @Bean

package com.edunex.backend.exception;

import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.authentication.BadCredentialsException;
import org.springframework.security.authentication.DisabledException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.multipart.MaxUploadSizeExceededException;

import java.util.HashMap;
import java.util.Map;

@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(BadCredentialsException.class)
    public ResponseEntity<Map<String, String>> handleBadCredentials(BadCredentialsException ex) {
        return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
                .body(Map.of("error", "Invalid email or password"));
    }

    @ExceptionHandler(DisabledException.class)
    public ResponseEntity<Map<String, String>> handleDisabled(DisabledException ex) {
        return ResponseEntity.status(HttpStatus.FORBIDDEN)
                .body(Map.of("error", "Account is not active"));
    }

    @ExceptionHandler(ConflictException.class)
    public ResponseEntity<Map<String, String>> handleConflict(ConflictException ex) {
        return ResponseEntity.status(HttpStatus.CONFLICT)
                .body(Map.of("error", ex.getMessage()));
    }

    @ExceptionHandler(ResourceNotFoundException.class)
    public ResponseEntity<Map<String, String>> handleNotFound(ResourceNotFoundException ex) {
        return ResponseEntity.status(HttpStatus.NOT_FOUND)
                .body(Map.of("error", ex.getMessage()));
    }

    @ExceptionHandler(BadRequestException.class)
    public ResponseEntity<Map<String, String>> handleBadRequest(BadRequestException ex) {
        return ResponseEntity.badRequest().body(Map.of("error", ex.getMessage()));
    }

    @ExceptionHandler(MaxUploadSizeExceededException.class)
    public ResponseEntity<Map<String, String>> handleTooLarge(MaxUploadSizeExceededException ex) {
        return ResponseEntity.status(HttpStatus.CONTENT_TOO_LARGE)
                .body(Map.of("error", "File is too large (25 MB maximum)"));
    }

    @ExceptionHandler(TooManyRequestsException.class)
    public ResponseEntity<Map<String, Object>> handleTooMany(TooManyRequestsException ex) {
        return ResponseEntity.status(HttpStatus.TOO_MANY_REQUESTS)
                .header("Retry-After", String.valueOf(ex.getRetryAfterSeconds()))
                .body(Map.<String, Object>of("error", ex.getMessage(), "retryAfterSeconds", ex.getRetryAfterSeconds()));
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<Map<String, Object>> handleValidation(MethodArgumentNotValidException ex) {
        Map<String, String> fields = new HashMap<>();
        ex.getBindingResult().getFieldErrors()
                .forEach(e -> fields.put(e.getField(), e.getDefaultMessage()));
        Map<String, Object> body = new HashMap<>();
        body.put("error", "Validation failed");
        body.put("fields", fields);
        return ResponseEntity.badRequest().body(body);
    }
    @ExceptionHandler(org.springframework.http.converter.HttpMessageNotReadableException.class)
    public ResponseEntity<Map<String, String>> handleUnreadable(
            org.springframework.http.converter.HttpMessageNotReadableException ex) {
        String msg = ex.getMostSpecificCause() != null
                ? ex.getMostSpecificCause().getMessage()
                : ex.getMessage();
        return ResponseEntity.badRequest().body(Map.of("error", "Malformed request body: " + msg));
    }

    @ExceptionHandler(jakarta.validation.ConstraintViolationException.class)
    public ResponseEntity<Map<String, String>> handleConstraintViolation(
            jakarta.validation.ConstraintViolationException ex) {
        return ResponseEntity.badRequest().body(Map.of("error", ex.getMessage()));
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<Map<String, String>> handleGeneric(Exception ex) {
        ex.printStackTrace();
        return ResponseEntity.internalServerError()
                .body(Map.of("error", ex.getClass().getSimpleName() + ": " + ex.getMessage()));
    }

}

package com.edunex.backend.controller;

import com.edunex.backend.dto.LoginRequest;
import com.edunex.backend.dto.LoginResponse;
import com.edunex.backend.entity.User;
import com.edunex.backend.exception.TooManyRequestsException;
import com.edunex.backend.repository.UserRepository;
import com.edunex.backend.security.JwtService;
import com.edunex.backend.security.LoginAttemptService;
import com.edunex.backend.service.ActivityLogService;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.http.ResponseEntity;
import org.springframework.security.authentication.AuthenticationManager;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.AuthenticationException;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

@RestController
@RequestMapping("/api/auth")
@RequiredArgsConstructor
public class AuthController {

    private final AuthenticationManager authenticationManager;
    private final UserRepository userRepository;
    private final JwtService jwtService;
    private final ActivityLogService activityLogService;
    private final LoginAttemptService loginAttempts;

    @PostMapping("/login")
    public ResponseEntity<LoginResponse> login(@Valid @RequestBody LoginRequest request, HttpServletRequest http) {
        String ip = http.getRemoteAddr();
        String key = request.email().toLowerCase() + "|" + ip;

        if (loginAttempts.isBlocked(key)) {
            throw new TooManyRequestsException("Too many failed attempts. Try again later.",
                    loginAttempts.retryAfterSeconds(key));
        }
        try {
            authenticationManager.authenticate(
                    new UsernamePasswordAuthenticationToken(request.email(), request.password()));
        } catch (AuthenticationException ex) {
            boolean locked = loginAttempts.recordFailure(key);
            activityLogService.registerForEmail(
                    (locked ? "LOGIN_LOCKED " : "LOGIN_FAILED ") + request.email(), "User", null, ip);
            throw ex;
        }
        loginAttempts.reset(key);

        User user = userRepository.findByEmail(request.email()).orElseThrow();
        String roleName = user.getRole() != null ? user.getRole().getName() : null;
        String token = jwtService.generateToken(user.getEmail(), roleName);
        activityLogService.registerForEmail("LOGIN", "User", user.getEmail(), ip);

        return ResponseEntity.ok(new LoginResponse(
                token, "Bearer", user.getId(), user.getEmail(),
                user.getFirstName(), user.getLastName(), roleName));
    }
}

import java.util.List;

@Service
@RequiredArgsConstructor
public class ActivityLogService {

    private final ActivityLogRepository logRepository;
    private final UserRepository userRepository;

    @Transactional
    public void registerForEmail(String action, String entityType, String email, String ip) {
        User user = email == null ? null : userRepository.findByEmail(email).orElse(null);
        ActivityLog log = new ActivityLog();
        log.setAction(action.length() > 200 ? action.substring(0, 200) : action);
        log.setEntityType(entityType);
        log.setIpAddress(ip);
        log.setUser(user);
        logRepository.save(log);
    }

    @Transactional(readOnly = true)
    public PageResponse<ActivityLogResponse> search(String q, int page, int size) {
        Pageable pageable = PageRequest.of(Math.max(page, 0), Math.min(Math.max(size, 1), 100));
        Page<ActivityLog> result = (q == null || q.isBlank())
                ? logRepository.findAllByOrderByTimestampDesc(pageable)
                : logRepository.search(q.trim(), pageable);
        List<ActivityLogResponse> items = result.getContent().stream().map(this::toResponse).toList();
        return new PageResponse<>(items, result.getNumber(), result.getSize(),
                result.getTotalElements(), result.getTotalPages());
    }

        User u = l.getUser();
        return new ActivityLogResponse(l.getId(), l.getAction(), l.getEntityType(), l.getTimestamp(),
                l.getIpAddress(), u != null ? u.getEmail() : null,
                u != null ? u.getFirstName() + " " + u.getLastName() : null);

package com.edunex.backend.service;

import com.edunex.backend.dto.CourseResponse;

import java.time.DayOfWeek;
import java.time.LocalDateTime;
import java.time.LocalTime;
import java.util.Comparator;

/** Composes the existing services, so every access rule still applies. */
@Service
@RequiredArgsConstructor
public class DashboardService {

    private final GradeService gradeService;
    private final AttendanceService attendanceService;
    private final TimetableService timetableService;
    private final CourseService courseService;
    private final NotificationService notificationService;
    private final ProfessorClassService professorClassService;

    public StudentDashboard student(String email) {
        StudentGradesSummary grades = gradeService.studentSummary(email);
        List<StudentAttendanceStats> attendance = attendanceService.studentStats(email);
        List<CourseResponse> courses = courseService.studentCourses(email);
        return new StudentDashboard(
                grades.subjects().size(),
                grades.overallAverage(),
                attendance.stream().mapToLong(StudentAttendanceStats::absent).sum(),
                attendance.stream().mapToLong(StudentAttendanceStats::late).sum(),
                notificationService.unreadCount(email),
                upcoming(timetableService.studentSessions(email)),
                courses.stream().limit(3)
                        .map(c -> new StudentDashboard.CourseBrief(c.id(), c.title(), c.subject().name(), c.createdAt()))
                        .toList());
    }

    public ProfessorDashboard professor(String email) {
        List<CourseResponse> courses = courseService.listMyCourses(email);
        List<ProfessorClassResponse> classes = professorClassService.myClasses(email);
        long students = classes.stream()
                .flatMap(c -> c.students().stream())
                .map(ProfessorClassResponse.StudentRef::id)
                .distinct()
                .count();
        return new ProfessorDashboard(
                courseService.professorSubjects(email).size(),
                courses.size(),
                courses.stream().mapToLong(c -> c.documents().size()).sum(),
                classes.size(),
                students,
                notificationService.unreadCount(email),
                upcoming(timetableService.professorSessions(email)));
    }

    private List<SessionResponse> upcoming(List<SessionResponse> sessions) {
        LocalDateTime now = LocalDateTime.now();
        DayOfWeek today = now.getDayOfWeek();
        LocalTime time = now.toLocalTime();
        return sessions.stream()
                .sorted(Comparator.comparingLong((SessionResponse s) -> minutesUntil(s, today, time)))
                .limit(3)
                .toList();
    }

    private static long minutesUntil(SessionResponse s, DayOfWeek today, LocalTime now) {
        int days = (DayOfWeek.valueOf(s.dayOfWeek()).getValue() - today.getValue() + 7) % 7;
        if (days == 0 && LocalTime.parse(s.endTime()).isBefore(now)) {
            days = 7;
        }
        return days * 24L * 60 + LocalTime.parse(s.startTime()).toSecondOfDay() / 60;
    }
}

package com.edunex.backend.service;

import com.edunex.backend.dto.CreateEnrollmentRequest;
import com.edunex.backend.dto.EnrollmentResponse;
import com.edunex.backend.entity.Enrollment;
import com.edunex.backend.entity.SchoolClass;
import com.edunex.backend.entity.Section;
import com.edunex.backend.entity.Student;
import com.edunex.backend.enums.EnrollmentStatus;
import com.edunex.backend.exception.ConflictException;
import com.edunex.backend.exception.ResourceNotFoundException;
import com.edunex.backend.repository.EnrollmentRepository;
import com.edunex.backend.repository.SchoolClassRepository;
import com.edunex.backend.repository.SectionRepository;
import com.edunex.backend.repository.StudentRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.util.Comparator;
import java.util.List;

@Service
@RequiredArgsConstructor
public class EnrollmentService {

    private final EnrollmentRepository enrollmentRepository;
    private final StudentRepository studentRepository;
    private final SchoolClassRepository classRepository;
    private final SectionRepository sectionRepository;

    @Transactional
    public EnrollmentResponse create(CreateEnrollmentRequest req) {
        Student student = studentRepository.findById(req.studentId())
                .orElseThrow(() -> new ResourceNotFoundException("Student not found"));
        SchoolClass schoolClass = classRepository.findById(req.classId())
                .orElseThrow(() -> new ResourceNotFoundException("Class not found"));

        Section section = null;
        if (req.sectionId() != null) {
            section = sectionRepository.findById(req.sectionId())
                    .orElseThrow(() -> new ResourceNotFoundException("Section not found"));
            if (section.getSchoolClass() == null
                    || !section.getSchoolClass().getId().equals(schoolClass.getId())) {
                throw new ConflictException("Section does not belong to this class");
            }
        }

        if (enrollmentRepository.existsByStudentIdAndSchoolClassIdAndAcademicYearAndStatus(
                student.getId(), schoolClass.getId(), schoolClass.getAcademicYear(), EnrollmentStatus.ACTIVE)) {
            throw new ConflictException("Student is already enrolled in this class for this academic year");
        }

        Enrollment e = new Enrollment();
        e.setStudent(student);
        e.setSchoolClass(schoolClass);
        e.setSection(section);
        e.setAcademicYear(schoolClass.getAcademicYear());
        e.setEnrollmentDate(req.enrollmentDate() != null ? req.enrollmentDate() : LocalDate.now());
        e.setStatus(EnrollmentStatus.ACTIVE);

        return toResponse(enrollmentRepository.save(e));
    }

    @Transactional(readOnly = true)
    public List<EnrollmentResponse> findAll(Long classId, Long studentId) {
        List<Enrollment> list;
        if (classId != null) {
            list = enrollmentRepository.findBySchoolClassId(classId);
        } else if (studentId != null) {
            list = enrollmentRepository.findByStudentId(studentId);
        } else {
            list = enrollmentRepository.findAll();
        }
        return list.stream()
                .sorted(Comparator.comparing(Enrollment::getId))
                .map(this::toResponse)
                .toList();
    }

    @Transactional
    public EnrollmentResponse updateStatus(Long id, EnrollmentStatus status) {
        Enrollment e = enrollmentRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Enrollment not found"));
        if (status == EnrollmentStatus.ACTIVE && e.getStatus() != EnrollmentStatus.ACTIVE
                && enrollmentRepository.existsByStudentIdAndSchoolClassIdAndAcademicYearAndStatus(
                        e.getStudent().getId(), e.getSchoolClass().getId(), e.getAcademicYear(), EnrollmentStatus.ACTIVE)) {
            throw new ConflictException("Student is already enrolled in this class for this academic year");
        }
        e.setStatus(status);
        return toResponse(enrollmentRepository.save(e));
    }

    private EnrollmentResponse toResponse(Enrollment e) {
        Student s = e.getStudent();
        SchoolClass c = e.getSchoolClass();
        Section sec = e.getSection();
        return new EnrollmentResponse(
                e.getId(),
                s.getId(), s.getFirstName() + " " + s.getLastName(), s.getStudentNumber(),
                c.getId(), c.getName(),
                sec != null ? sec.getId() : null,
                sec != null ? sec.getName() : null,
                e.getAcademicYear(), e.getEnrollmentDate(), e.getStatus().name());
    }
}

package com.edunex.backend.service;

import com.edunex.backend.exception.ResourceNotFoundException;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.io.FileSystemResource;

import org.springframework.stereotype.Component;
import org.springframework.web.multipart.MultipartFile;

import java.io.IOException;
import java.io.InputStream;
import java.io.UncheckedIOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardCopyOption;
import java.util.UUID;

/** Development storage on the local disk. Replace with an object storage implementation (R2, Cloudinary) for production. */
@Slf4j
@Component
public class LocalFileStorage implements FileStorage {

    private final Path root;

    public LocalFileStorage(@Value("${app.storage.path:./storage}") String path) {
        this.root = Paths.get(path).toAbsolutePath().normalize();
    }

    @Override
    public String store(MultipartFile file) {
        String original = file.getOriginalFilename() == null ? "file" : file.getOriginalFilename();
        String name = original.substring(Math.max(original.lastIndexOf('/'), original.lastIndexOf('\\')) + 1);
        String safeName = name.replaceAll("[^A-Za-z0-9._-]", "_");
        if (safeName.isBlank()) {
            safeName = "file";
        }
        String key = UUID.randomUUID() + "/" + safeName;
        Path target = resolve(key);
        try {
            Files.createDirectories(target.getParent());
            try (InputStream in = file.getInputStream()) {
                Files.copy(in, target, StandardCopyOption.REPLACE_EXISTING);
            }
        } catch (IOException e) {
            throw new UncheckedIOException("Could not store file", e);
        }
        return key;
    }

    @Override
    public Resource load(String key) {
        Path path = resolve(key);
        if (!Files.exists(path)) {
            throw new ResourceNotFoundException("File not found");
        }
        return new FileSystemResource(path);
    }

    @Override
    public void delete(String key) {
        try {
            Path path = resolve(key);
            Files.deleteIfExists(path);
            Path dir = path.getParent();
            if (dir != null && !dir.equals(root)) {
                Files.deleteIfExists(dir);
            }
        } catch (IOException e) {
            log.warn("Could not delete stored file {}", key, e);
        }
    }

    private Path resolve(String key) {
        Path path = root.resolve(key).normalize();
        if (!path.startsWith(root)) {
            throw new IllegalArgumentException("Invalid file key");
        }
        return path;
    }
}

    private final SubjectRepository subjectRepository;
    private final SchoolClassRepository classRepository;
    private final EnrollmentRepository enrollmentRepository;

    @Transactional(readOnly = true)
    public List<ProfessorClassResponse> myClasses(String email) {
        Professor p = userRepository.findByEmail(email)
                .filter(u -> u instanceof Professor)
                .map(u -> (Professor) u)
                .orElseThrow(() -> new AccessDeniedException("Professor account required"));

        Map<Long, SchoolClass> classes = new LinkedHashMap<>();
        Map<Long, List<Subject>> subjectsByClass = new HashMap<>();
        for (Subject s : subjectRepository.findByProfessorsId(p.getId())) {
            for (SchoolClass c : classRepository.findBySubjectsId(s.getId())) {
                classes.putIfAbsent(c.getId(), c);
                subjectsByClass.computeIfAbsent(c.getId(), k -> new ArrayList<>()).add(s);
            }
        }
        if (classes.isEmpty()) {
            return List.of();
        }

        Map<Long, List<Enrollment>> enrollments = enrollmentRepository
                .findBySchoolClassIdInAndStatus(classes.keySet(), EnrollmentStatus.ACTIVE).stream()
                .collect(Collectors.groupingBy(e -> e.getSchoolClass().getId()));

        return classes.values().stream()
                .sorted(Comparator.comparing(SchoolClass::getName))
                .map(c -> new ProfessorClassResponse(
                        c.getId(), c.getName(), c.getLevel(), c.getAcademicYear(),
                        subjectsByClass.get(c.getId()).stream()
                                .sorted(Comparator.comparing(Subject::getName))
                                .map(s -> new ProfessorClassResponse.SubjectRef(s.getId(), s.getName(), s.getCode()))
                                .toList(),
                        enrollments.getOrDefault(c.getId(), List.of()).stream()
                                .sorted(Comparator.comparing((Enrollment e) -> e.getStudent().getLastName())
                                        .thenComparing(e -> e.getStudent().getFirstName()))

                                        e.getSection() != null ? e.getSection().getName() : null))

@Service
@RequiredArgsConstructor
public class ProfileService {

    private static final Set<String> PHOTO_TYPES = Set.of("png", "jpg", "jpeg", "webp");
    private static final long MAX_PHOTO_BYTES = 2L * 1024 * 1024;

    private final UserRepository userRepository;
    private final ProfileRepository profileRepository;
    private final PasswordEncoder passwordEncoder;
    private final FileStorage fileStorage;

    public record PhotoFile(Resource resource, String filename) {
    }

    @Transactional(readOnly = true)
    public ProfileResponse get(String email) {
        User u = user(email);
        return toResponse(u, profileRepository.findByUserId(u.getId()).orElse(null));
    }

    @Transactional
    public ProfileResponse update(String email, ProfileUpdateRequest req) {
        User u = user(email);
        u.setPhone(blankToNull(req.phone()));
        userRepository.save(u);
        Profile p = profileOf(u);
        p.setAddress(blankToNull(req.address()));
        p.setDateOfBirth(req.dateOfBirth());
        return toResponse(u, profileRepository.save(p));
    }

    @Transactional
    public void changePassword(String email, ChangePasswordRequest req) {
        User u = user(email);
        if (!passwordEncoder.matches(req.currentPassword(), u.getPassword())) {
            throw new BadRequestException("Current password is incorrect");
        }
        if (passwordEncoder.matches(req.newPassword(), u.getPassword())) {
            throw new BadRequestException("New password must be different from the current one");
        }
        u.setPassword(passwordEncoder.encode(req.newPassword()));
        userRepository.save(u);
    }

    @Transactional
    public ProfileResponse uploadPhoto(String email, MultipartFile file) {
        if (file == null || file.isEmpty()) {
            throw new BadRequestException("File is empty");
        }
        if (file.getSize() > MAX_PHOTO_BYTES) {
            throw new BadRequestException("Photo is too large (2 MB maximum)");
        }
        String name = file.getOriginalFilename() == null ? "" : file.getOriginalFilename();
        int dot = name.lastIndexOf('.');
        String ext = dot < 0 ? "" : name.substring(dot + 1).toLowerCase();
        if (!PHOTO_TYPES.contains(ext)) {
            throw new BadRequestException("Photo must be a PNG, JPG or WEBP image");
        }
        User u = user(email);
        Profile p = profileOf(u);
        String oldKey = p.getPhotoUrl();
        p.setPhotoUrl(fileStorage.store(file));
        Profile saved = profileRepository.save(p);
        if (oldKey != null) {
            fileStorage.delete(oldKey);
        }
        return toResponse(u, saved);
    }

    @Transactional(readOnly = true)
    public PhotoFile photo(String email) {
        User u = user(email);
        Profile p = profileRepository.findByUserId(u.getId())
                .filter(x -> x.getPhotoUrl() != null)
                .orElseThrow(() -> new ResourceNotFoundException("No photo"));
        return new PhotoFile(fileStorage.load(p.getPhotoUrl()), Paths.get(p.getPhotoUrl()).getFileName().toString());

            fileStorage.delete(p.getPhotoUrl());
            p.setPhotoUrl(null);
            p = profileRepository.save(p);
        }
        return toResponse(u, p);
    }

    }

        });
    }

        }

    private final SchoolClassRepository classRepository;
    private final SubjectRepository subjectRepository;
    private final CourseRepository courseRepository;
    private final CourseDocumentRepository documentRepository;

    private final GradeRepository gradeRepository;
    private final UserRepository userRepository;
    private final RoleRepository roleRepository;
    private final ActivityLogRepository logRepository;
    private final PlatformSettingRepository settingRepository;

    @Transactional(readOnly = true)
    public UniversityStats university() {
        UniversityStats.Counts counts = new UniversityStats.Counts(
                studentRepository.count(),
                scalar("select count(s) from Student s where s.status = :st", UserStatus.ACTIVE),
                professorRepository.count(),
                classRepository.count(),
                subjectRepository.count(),
                scalar("select count(e) from Enrollment e where e.status = :st", EnrollmentStatus.ACTIVE),
                courseRepository.count(),
                documentRepository.count(),
                sessionRepository.count());

        List<UniversityStats.ClassCount> byClass = em.createQuery(
                        "select c.name, count(e) from Enrollment e join e.schoolClass c "
                                + "where e.status = :st group by c.name order by c.name", Object[].class)
                .setParameter("st", EnrollmentStatus.ACTIVE)
                .getResultList().stream()
                .map(r -> new UniversityStats.ClassCount((String) r[0], (Long) r[1]))
                .toList();

        return new UniversityStats(counts, byClass, attendance(), grades());
    }

    private UniversityStats.AttendanceStats attendance() {
        long present = 0;
        long absent = 0;
        long late = 0;
        for (Object[] r : em.createQuery("select a.status, count(a) from Attendance a group by a.status", Object[].class)
                .getResultList()) {
            long n = (Long) r[1];
            switch ((AttendanceStatus) r[0]) {
                case PRESENT -> present = n;
                case ABSENT -> absent = n;
                case LATE -> late = n;
            }
        }
        long total = present + absent + late;

        Map<String, long[]> bySubject = new TreeMap<>();
        for (Object[] r : em.createQuery(
                        "select s.name, a.status, count(a) from Attendance a join a.subject s group by s.name, a.status",
                        Object[].class).getResultList()) {
            long[] totals = bySubject.computeIfAbsent((String) r[0], k -> new long[2]);
            long n = (Long) r[2];
            totals[1] += n;
            if (r[1] == AttendanceStatus.ABSENT) {
                totals[0] += n;
            }
        }
        List<UniversityStats.SubjectAbsence> subjects = bySubject.entrySet().stream()
                .map(e -> new UniversityStats.SubjectAbsence(e.getKey(), e.getValue()[0], e.getValue()[1],
                        rate(e.getValue()[0], e.getValue()[1])))
                .toList();
        return new UniversityStats.AttendanceStats(present, absent, late, total, rate(absent, total), subjects);
    }

    private UniversityStats.GradeStats grades() {
        List<Grade> grades = gradeRepository.findAll();
        long[] buckets = new long[4];
        double sum = 0;
        for (Grade g : grades) {
            double v = g.getValue();
            sum += v;
            buckets[v < 5 ? 0 : v < 10 ? 1 : v < 15 ? 2 : 3]++;
        }
        Double average = grades.isEmpty() ? null : round2(sum / grades.size());
        List<UniversityStats.Bucket> distribution = List.of(
                new UniversityStats.Bucket("0-5", buckets[0]),
                new UniversityStats.Bucket("5-10", buckets[1]),
                new UniversityStats.Bucket("10-15", buckets[2]),
                new UniversityStats.Bucket("15-20", buckets[3]));

        Map<String, DoubleSummaryStatistics> bySubject = grades.stream().collect(Collectors.groupingBy(
                g -> g.getSubject().getName(), TreeMap::new, Collectors.summarizingDouble(Grade::getValue)));
        List<UniversityStats.SubjectAverage> subjects = bySubject.entrySet().stream()
                .map(e -> new UniversityStats.SubjectAverage(e.getKey(), round2(e.getValue().getAverage()), e.getValue().getCount()))
                .toList();
        return new UniversityStats.GradeStats(grades.size(), average, distribution, subjects);
    }

    @Transactional(readOnly = true)
    public PlatformStats platform() {
        List<PlatformStats.RoleCount> byRole = roleRepository.findAll().stream()
                .sorted(Comparator.comparing(Role::getName))
                .map(r -> new PlatformStats.RoleCount(r.getName(), userRepository.countByRoleId(r.getId())))
                .toList();
        List<PlatformStats.StatusCount> byStatus = Arrays.stream(UserStatus.values())
                .map(s -> new PlatformStats.StatusCount(s.name(), userRepository.countByStatus(s)))
                .toList();

        LocalDate today = LocalDate.now();
        Map<LocalDate, long[]> days = new TreeMap<>();
        for (int i = 6; i >= 0; i--) {
            days.put(today.minusDays(i), new long[3]);
        }
        for (ActivityLog l : logRepository.findByTimestampAfter(today.minusDays(6).atStartOfDay())) {
            long[] d = days.get(l.getTimestamp().toLocalDate());
            if (d == null) {
                continue;
            }
            String a = l.getAction();
            if (a.equals("LOGIN")) {
                d[0]++;
            } else if (a.startsWith("LOGIN_FAILED") || a.startsWith("LOGIN_LOCKED")) {
                d[1]++;
            } else if (a.startsWith("CREATE ") || a.startsWith("UPDATE ") || a.startsWith("DELETE ")) {
                d[2]++;
            }
        }
        List<PlatformStats.DayActivity> last7 = days.entrySet().stream()
                .map(e -> new PlatformStats.DayActivity(e.getKey().toString(), e.getValue()[0], e.getValue()[1], e.getValue()[2]))
                .toList();

        return new PlatformStats(userRepository.count(), byRole, byStatus,
                logRepository.count(), settingRepository.count(), last7);
    }

    private long scalar(String jpql, Object status) {
        return em.createQuery(jpql, Long.class).setParameter("st", status).getSingleResult();
    }

    private static double rate(long part, long total) {
        return total == 0 ? 0 : Math.round(part * 1000.0 / total) / 10.0;
    }

    private static double round2(double v) {
        return Math.round(v * 100.0) / 100.0;
    }
}

package com.edunex.backend.service;

import com.edunex.backend.dto.SessionRequest;
import com.edunex.backend.dto.SessionResponse;
import com.edunex.backend.dto.TimetableRequest;
import com.edunex.backend.dto.TimetableResponse;
import com.edunex.backend.entity.Enrollment;
import com.edunex.backend.entity.Professor;
import com.edunex.backend.entity.SchoolClass;
import com.edunex.backend.entity.Student;
import com.edunex.backend.entity.Subject;
import com.edunex.backend.entity.Timetable;
import com.edunex.backend.entity.TimetableSession;
import com.edunex.backend.enums.EnrollmentStatus;
import com.edunex.backend.enums.NotificationType;
import com.edunex.backend.exception.BadRequestException;
import com.edunex.backend.exception.ConflictException;
import com.edunex.backend.exception.ResourceNotFoundException;
import com.edunex.backend.repository.EnrollmentRepository;
import com.edunex.backend.repository.ProfessorRepository;
import com.edunex.backend.repository.SchoolClassRepository;
import com.edunex.backend.repository.SubjectRepository;
import com.edunex.backend.repository.TimetableRepository;
import com.edunex.backend.repository.TimetableSessionRepository;
import com.edunex.backend.repository.UserRepository;
import lombok.RequiredArgsConstructor;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.Comparator;
import java.util.List;
import java.util.Set;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
public class TimetableService {

    private static final Comparator<TimetableSession> ORDER =
            Comparator.comparing((TimetableSession s) -> s.getDayOfWeek().getValue())
                    .thenComparing(TimetableSession::getStartTime);

    private final TimetableRepository timetableRepository;
    private final TimetableSessionRepository sessionRepository;
    private final SchoolClassRepository classRepository;
    private final SubjectRepository subjectRepository;
    private final ProfessorRepository professorRepository;
    private final EnrollmentRepository enrollmentRepository;
    private final UserRepository userRepository;
    private final NotificationService notificationService;

    // ---------- administration agent ----------

    @Transactional
    public TimetableResponse create(TimetableRequest req) {
        String year = req.academicYear().trim();
        String semester = req.semester().trim();
        if (timetableRepository.findByAcademicYearAndSemester(year, semester).isPresent()) {
            throw new ConflictException("A timetable already exists for this academic year and semester");
        }
        Timetable t = new Timetable();
        t.setAcademicYear(year);
        t.setSemester(semester);
        return toResponse(timetableRepository.save(t));
    }

    @Transactional(readOnly = true)
    public List<TimetableResponse> findAll() {
        return timetableRepository.findAll().stream()
                .sorted(Comparator.comparing(Timetable::getAcademicYear).thenComparing(Timetable::getSemester))
                .map(this::toResponse)
                .toList();
    }

    @Transactional(readOnly = true)
    public List<SessionResponse> sessions(Long timetableId) {
        findTimetable(timetableId);
        return sessionRepository.findByTimetableId(timetableId).stream()
                .sorted(ORDER)
                .map(this::toResponse)
                .toList();
    }

    @Transactional
    public SessionResponse addSession(Long timetableId, SessionRequest req) {
        Timetable t = findTimetable(timetableId);
        TimetableSession s = new TimetableSession();
        s.setTimetable(t);
        apply(s, t, req, null);
        notifySession("Nouvelle séance", s);
        return toResponse(sessionRepository.save(s));
    }

    @Transactional
    public SessionResponse updateSession(Long sessionId, SessionRequest req) {
        TimetableSession s = findSession(sessionId);
        apply(s, s.getTimetable(), req, sessionId);
        notifySession("Séance modifiée", s);
        return toResponse(sessionRepository.save(s));
    }

    @Transactional
    public void deleteSession(Long sessionId) {
        TimetableSession s = findSession(sessionId);
        notifySession("Séance annulée", s);
        s.getTimetable().getSessions().remove(s);
        sessionRepository.delete(s);
    }

    // ---------- professor and student views ----------

    @Transactional(readOnly = true)
    public List<SessionResponse> professorSessions(String email) {
        Professor p = userRepository.findByEmail(email)
                .filter(u -> u instanceof Professor)
                .map(u -> (Professor) u)
                .orElseThrow(() -> new AccessDeniedException("Professor account required"));
        return sessionRepository.findByProfessorId(p.getId()).stream()
                .sorted(ORDER)
                .map(this::toResponse)
                .toList();
    }

    @Transactional(readOnly = true)
    public List<SessionResponse> studentSessions(String email) {
        Student st = userRepository.findByEmail(email)
                .filter(u -> u instanceof Student)
                .map(u -> (Student) u)
                .orElseThrow(() -> new AccessDeniedException("Student account required"));
        Set<Long> classIds = enrollmentRepository.findByStudentIdAndStatus(st.getId(), EnrollmentStatus.ACTIVE)
                .stream()
                .map(Enrollment::getSchoolClass)
                .map(SchoolClass::getId)
                .collect(Collectors.toSet());
        if (classIds.isEmpty()) {
            return List.of();
        }
        return sessionRepository.findBySchoolClassIdIn(classIds).stream()
                .sorted(ORDER)
                .map(this::toResponse)
                .toList();
    }

    // ---------- rules ----------

    private void apply(TimetableSession s, Timetable t, SessionRequest req, Long excludeId) {
        if (!req.startTime().isBefore(req.endTime())) {
            throw new BadRequestException("End time must be after start time");
        }
        SchoolClass c = classRepository.findById(req.classId())
                .orElseThrow(() -> new ResourceNotFoundException("Class not found"));
        Subject subject = subjectRepository.findById(req.subjectId())
                .orElseThrow(() -> new ResourceNotFoundException("Subject not found"));
        Professor prof = professorRepository.findById(req.professorId())
                .orElseThrow(() -> new ResourceNotFoundException("Professor not found"));

        if (c.getSubjects().stream().noneMatch(x -> x.getId().equals(subject.getId()))) {
            throw new BadRequestException("This subject is not taught in this class");
        }
        if (subject.getProfessors().stream().noneMatch(x -> x.getId().equals(prof.getId()))) {
            throw new BadRequestException("This professor is not assigned to this subject");
        }

        String room = req.room() == null || req.room().isBlank() ? null : req.room().trim();

        for (TimetableSession other : sessionRepository.findByTimetableIdAndDayOfWeek(t.getId(), req.dayOfWeek())) {
            if (excludeId != null && other.getId().equals(excludeId)) {
                continue;
            }
            boolean overlap = req.startTime().isBefore(other.getEndTime())
                    && req.endTime().isAfter(other.getStartTime());
            if (!overlap) {
                continue;
            }
            if (other.getProfessor().getId().equals(prof.getId())) {
                throw new ConflictException("Conflict: the professor already has a session at this time");
            }
            if (room != null && other.getRoom() != null && other.getRoom().equalsIgnoreCase(room)) {
                throw new ConflictException("Conflict: the room is already used at this time");
            }
            if (other.getSchoolClass().getId().equals(c.getId())) {
                throw new ConflictException("Conflict: the class already has a session at this time");
            }
        }

        s.setDayOfWeek(req.dayOfWeek());
        s.setStartTime(req.startTime());
        s.setEndTime(req.endTime());
        s.setRoom(room);
        s.setSchoolClass(c);
        s.setSubject(subject);
        s.setProfessor(prof);
    }

    // ---------- helpers ----------

    private void notifySession(String prefix, TimetableSession s) {
        String day = s.getDayOfWeek().getDisplayName(java.time.format.TextStyle.FULL, java.util.Locale.FRENCH);
        String msg = prefix + " : " + s.getSubject().getName() + ", " + day + " de " + s.getStartTime()
                + " à " + s.getEndTime() + (s.getRoom() != null ? " (salle " + s.getRoom() + ")" : "");
        notificationService.notifyClassStudents(s.getSchoolClass().getId(), NotificationType.TIMETABLE_CHANGE, msg);
        notificationService.notifyUsers(java.util.List.of(s.getProfessor()), NotificationType.TIMETABLE_CHANGE, msg);
    }

    private Timetable findTimetable(Long id) {
        return timetableRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Timetable not found"));
    }

    private TimetableSession findSession(Long id) {
        return sessionRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Session not found"));
    }

    private TimetableResponse toResponse(Timetable t) {
        return new TimetableResponse(t.getId(), t.getAcademicYear(), t.getSemester(), t.getSessions().size());
    }

    private SessionResponse toResponse(TimetableSession s) {
        Timetable t = s.getTimetable();
        Professor p = s.getProfessor();
        return new SessionResponse(
                s.getId(), t.getId(), t.getAcademicYear(), t.getSemester(),
                s.getDayOfWeek().name(), s.getStartTime().toString(), s.getEndTime().toString(), s.getRoom(),
                s.getSchoolClass().getId(), s.getSchoolClass().getName(),
                s.getSubject().getId(), s.getSubject().getName(), s.getSubject().getCode(),
                p.getId(), p.getFirstName() + " " + p.getLastName());
    }
}

import { clearSession, getToken } from "./auth";

export class ApiError extends Error {
  status: number;
  constructor(status: number, message: string) {
    super(message);
    this.status = status;
  }
}

function authHeader(): Record<string, string> {
  const token = getToken();
  return token ? { Authorization: `Bearer ${token}` } : {};
}

function handleUnauthorized() {
  clearSession();
  if (typeof window !== "undefined") window.location.href = "/login";
}

export async function apiFetch<T>(path: string, init: RequestInit = {}): Promise<T> {
  const isForm = typeof FormData !== "undefined" && init.body instanceof FormData;
  let res: Response;
  try {
    res = await fetch(path, {
      ...init,
      headers: {
        ...(isForm ? {} : { "Content-Type": "application/json" }),
        ...authHeader(),
        ...(init.headers ?? {}),
      },
    });
  } catch {
    throw new ApiError(0, "Serveur injoignable. Vérifiez que le backend est démarré.");
  }

  if (res.status === 401) {
    handleUnauthorized();
    throw new ApiError(401, "Session expirée.");
  }

  if (!res.ok) {
    let message = "Une erreur est survenue.";
    try {
      const data = await res.json();
      if (res.status === 403) message = "Accès refusé.";
      else if (data?.fields)
        message = Object.entries(data.fields).map(([k, v]) => `${k} : ${v}`).join(", ");
      else if (data?.error) message = data.error;
    } catch {
      /* no JSON body */
    }
    throw new ApiError(res.status, message);
  }

  return res.status === 204 ? (undefined as T) : res.json();
}

export async function downloadDocument(id: number, fallbackName: string) {
  let res: Response;
  try {
    res = await fetch(`/api/documents/${id}/download`, { headers: authHeader() });
  } catch {
    throw new ApiError(0, "Serveur injoignable.");
  }
  if (res.status === 401) {
    handleUnauthorized();
    throw new ApiError(401, "Session expirée.");
  }
  if (!res.ok) {
    throw new ApiError(res.status, res.status === 403 ? "Accès refusé." : "Téléchargement impossible.");
  }

  const blob = await res.blob();
  const disposition = res.headers.get("Content-Disposition") ?? "";
  const match = /filename="?([^";]+)"?/i.exec(disposition);
  const name = match ? decodeURIComponent(match[1]) : fallbackName;

  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = name;
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(url);
}

  useEffect(() => {
    let active = true;
    const load = () =>
      apiFetch<{ count: number }>("/api/notifications/unread-count")
        .then((r) => {

    };

    <Link
      href={href}
      aria-label="Notifications"

import { ReactNode, createContext, useContext, useEffect, useState } from "react";
import Link from "next/link";
import { usePathname, useRouter } from "next/navigation";
import { apiFetch } from "@/lib/api";
import { clearSession } from "@/lib/auth";
import { homeFor, notificationsFor, profileFor } from "@/lib/roles";
import { Logo } from "./Logo";
import { NotificationBell } from "./NotificationBell";

export type Me = {
  id: number;
  email: string;
  firstName: string;
  lastName: string;
  phone: string | null;
  role: string | null;
  status: string;
};

export type NavItem = { href: string; label: string; soon?: boolean; exact?: boolean };

const MeContext = createContext<Me | null>(null);

export function useMe(): Me {
  const me = useContext(MeContext);
  if (!me) throw new Error("useMe must be used inside RoleShell");
  return me;
}

// Kept between page navigations so the layout does not flash "Chargement..." each time
let cachedMe: Me | null = null;

type Props = {
  allowedRole: string;
  badge: string;
  badgeClass: string;
  nav: NavItem[];
  children: ReactNode;
};

export function RoleShell({ allowedRole, badge, badgeClass, nav, children }: Props) {
  const router = useRouter();
  const pathname = usePathname();
  const [me, setMe] = useState<Me | null>(cachedMe?.role === allowedRole ? cachedMe : null);

  useEffect(() => {
    apiFetch<Me>("/api/users/me")
      .then((data) => {
        if (data.role === allowedRole) {
          cachedMe = data;
          setMe(data);
        } else {
          router.replace(homeFor(data.role));
        }
      })
      .catch(() => {});
  }, [allowedRole, router]);

  function logout() {
    cachedMe = null;
    clearSession();
    router.replace("/login");
  }

  if (!me) return <p className="p-10 text-slate-500">Chargement...</p>;

  const isActive = (item: NavItem) =>
    item.exact ? pathname === item.href : pathname === item.href || pathname.startsWith(item.href + "/");

  const soonTag = (
    <span className="rounded-full bg-slate-100 px-2 py-0.5 text-[10px] font-semibold uppercase text-slate-500">
      Bientôt
    </span>
  );

  return (
    <MeContext.Provider value={me}>
      <div className="min-h-screen bg-slate-50">
        <header className="flex flex-wrap items-center justify-between gap-4 border-b border-slate-200 bg-white px-6 py-4">
          <div className="flex flex-wrap items-center gap-4">
            <Logo />
            <span className={`rounded-full px-3 py-1 text-xs font-semibold ${badgeClass}`}>{badge}</span>
          </div>
          <div className="flex items-center gap-4 text-sm">
            <NotificationBell href={notificationsFor(me.role)} />
            <Link href={profileFor(me.role)} className="text-slate-600 hover:text-blue-700">
              {me.firstName} {me.lastName}
            </Link>
            <button
              onClick={logout}
              className="rounded-xl border border-slate-200 px-4 py-2 font-medium text-slate-700 hover:bg-slate-50"
            >
              Se déconnecter
            </button>
          </div>
        </header>

        {/* Mobile menu */}
        <nav className="flex gap-2 overflow-x-auto border-b border-slate-200 bg-white px-4 py-2 md:hidden">
          {nav.filter((i) => !i.soon).map((item) => (
            <Link
              key={item.href}
              href={item.href}
              className={`whitespace-nowrap rounded-lg px-3 py-2 text-sm font-medium ${
                isActive(item) ? "bg-blue-50 text-blue-700" : "text-slate-600"
              }`}
            >
              {item.label}
            </Link>
          ))}
        </nav>

        <div className="flex">
          <nav className="hidden min-h-[calc(100vh-73px)] w-64 shrink-0 border-r border-slate-200 bg-white p-4 md:block">
            <ul className="space-y-1">
              {nav.map((item) => (
                <li key={item.label}>
                  {item.soon ? (
                    <span className="flex cursor-not-allowed items-center justify-between rounded-xl px-4 py-3 text-sm text-slate-400">
                      {item.label}
                      {soonTag}
                    </span>
                  ) : (
                    <Link
                      href={item.href}
                      className={`block rounded-xl px-4 py-3 text-sm font-medium ${
                        isActive(item) ? "bg-blue-50 text-blue-700" : "text-slate-600 hover:bg-slate-50"
                      }`}
                    >
                      {item.label}
                    </Link>
                  )}
                </li>
              ))}
            </ul>
          </nav>
          <main className="min-w-0 flex-1 p-6">{children}</main>
        </div>
      </div>
    </MeContext.Provider>
  );
}

import { FormEvent, Fragment, useCallback, useEffect, useState } from "react";
import { AdminShell } from "@/components/AdminShell";
import { ApiError, apiFetch } from "@/lib/api";
import {
  ErrorBox,
  IdPicker,
  PageHeader,
  TextField,
  inputCls,
  primaryBtn,
  smallBtn,
  tableWrap,
  theadCls,
} from "@/components/ui";

type SubjectRef = { id: number; name: string; code: string };
type ClassItem = { id: number; name: string; level: string | null; academicYear: string; subjects: SubjectRef[] };
type Section = { id: number; name: string; classId: number };

const emptyForm = { name: "", level: "", academicYear: "" };

export default function ClassesPage() {
  const [classes, setClasses] = useState<ClassItem[]>([]);
  const [subjects, setSubjects] = useState<SubjectRef[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [showForm, setShowForm] = useState(false);
  const [form, setForm] = useState(emptyForm);
  const [saving, setSaving] = useState(false);
  const [openId, setOpenId] = useState<number | null>(null);
  const [mode, setMode] = useState<"subjects" | "sections">("subjects");
  const [sections, setSections] = useState<Section[]>([]);
  const [newSection, setNewSection] = useState("");

  const load = useCallback(async () => {
    try {
      const [c, s] = await Promise.all([
        apiFetch<ClassItem[]>("/api/admin/classes"),
        apiFetch<SubjectRef[]>("/api/admin/subjects"),
      ]);
      setClasses(c);
      setSubjects(s);
      setError(null);
    } catch (e) {
      if (e instanceof ApiError && e.status !== 401) setError(e.message);
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => {
    load();
  }, [load]);

  async function onCreate(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setSaving(true);
    setError(null);
    try {
      const created = await apiFetch<ClassItem>("/api/admin/classes", {
        method: "POST",
        body: JSON.stringify({
          name: form.name.trim(),
          level: form.level.trim() || null,
          academicYear: form.academicYear.trim(),
        }),
      });
      setClasses((prev) => [...prev, created].sort((a, b) => a.name.localeCompare(b.name)));
      setForm(emptyForm);
      setShowForm(false);
    } catch (err) {
      if (err instanceof ApiError) setError(err.message);
    } finally {
      setSaving(false);
    }
  }

  async function openPanel(id: number, m: "subjects" | "sections") {
    if (openId === id && mode === m) {
      setOpenId(null);
      return;
    }
    setOpenId(id);
    setMode(m);
    setError(null);
    setNewSection("");
    if (m === "sections") {
      try {
        setSections(await apiFetch<Section[]>(`/api/admin/classes/${id}/sections`));
      } catch (err) {
        if (err instanceof ApiError) setError(err.message);
      }
    }
  }

  async function assignSubjects(id: number, ids: number[]) {
    setSaving(true);
    setError(null);
    try {
      const updated = await apiFetch<ClassItem>(`/api/admin/classes/${id}/subjects`, {
        method: "PUT",
        body: JSON.stringify({ ids }),
      });
      setClasses((prev) => prev.map((c) => (c.id === id ? updated : c)));
      setOpenId(null);
    } catch (err) {
      if (err instanceof ApiError) setError(err.message);
    } finally {
      setSaving(false);
    }
  }

  async function addSection(classId: number) {
    if (!newSection.trim()) return;
    setSaving(true);
    setError(null);
    try {
      const created = await apiFetch<Section>(`/api/admin/classes/${classId}/sections`, {
        method: "POST",
        body: JSON.stringify({ name: newSection.trim() }),
      });
      setSections((prev) => [...prev, created]);
      setNewSection("");
    } catch (err) {
      if (err instanceof ApiError) setError(err.message);
    } finally {
      setSaving(false);
    }
  }

  return (
    <AdminShell>
      <PageHeader
        title="Classes"
        subtitle="Créez les classes, rattachez-leur des matières et gérez leurs sections."
        action={
          <button onClick={() => setShowForm((v) => !v)} className={primaryBtn}>
            {showForm ? "Fermer" : "+ Nouvelle classe"}
          </button>
        }
      />
      <ErrorBox message={error} />

      {showForm && (
        <form onSubmit={onCreate} className="mt-6 rounded-2xl border border-slate-200 bg-white p-6 shadow-sm">
          <div className="grid gap-4 sm:grid-cols-3">
            <TextField label="Nom" value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} required />
            <TextField label="Niveau" value={form.level} onChange={(e) => setForm({ ...form, level: e.target.value })} />
            <TextField label="Année universitaire (ex. 2026-2027)" value={form.academicYear} onChange={(e) => setForm({ ...form, academicYear: e.target.value })} required />
          </div>
          <button type="submit" disabled={saving} className={`mt-6 ${primaryBtn}`}>
            {saving ? "Création..." : "Créer la classe"}
          </button>
        </form>
      )}

      <div className={tableWrap}>
        <table className="w-full text-left text-sm">
          <thead className={theadCls}>
            <tr>
              <th className="px-5 py-3">Classe</th>
              <th className="px-5 py-3">Niveau</th>
              <th className="px-5 py-3">Année</th>
              <th className="px-5 py-3">Matières</th>
              <th className="px-5 py-3 text-right">Actions</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-slate-100">
            {loading && (
              <tr><td colSpan={5} className="px-5 py-8 text-center text-slate-500">Chargement...</td></tr>
            )}
            {!loading && classes.length === 0 && (
              <tr><td colSpan={5} className="px-5 py-8 text-center text-slate-500">Aucune classe pour le moment.</td></tr>
            )}
            {classes.map((c) => (
              <Fragment key={c.id}>
                <tr>
                  <td className="px-5 py-4 font-medium text-slate-900">{c.name}</td>
                  <td className="px-5 py-4 text-slate-600">{c.level ?? "-"}</td>
                  <td className="px-5 py-4 text-slate-600">{c.academicYear}</td>
                  <td className="px-5 py-4 text-slate-600">
                    {c.subjects.length ? c.subjects.map((s) => s.code).join(", ") : "-"}
                  </td>
                  <td className="px-5 py-4 text-right">
                    <div className="flex justify-end gap-2">
                      <button onClick={() => openPanel(c.id, "subjects")} className={smallBtn}>Matières</button>
                      <button onClick={() => openPanel(c.id, "sections")} className={smallBtn}>Sections</button>
                    </div>
                  </td>
                </tr>
                {openId === c.id && (
                  <tr className="bg-slate-50">
                    <td colSpan={5} className="px-5 py-5">
                      {mode === "subjects" ? (
                        <>
                          <p className="mb-3 text-sm font-medium">Matières de {c.name}</p>
                          <IdPicker
                            options={subjects.map((s) => ({ id: s.id, label: `${s.name} (${s.code})` }))}
                            selected={c.subjects.map((s) => s.id)}
                            onSave={(ids) => assignSubjects(c.id, ids)}
                            saving={saving}
                            emptyText="Aucune matière. Créez-en d&apos;abord dans la page Matières."
                          />
                        </>
                      ) : (
                        <>
                          <p className="mb-3 text-sm font-medium">Sections de {c.name}</p>
                          <div className="flex flex-wrap gap-2">
                            {sections.length === 0 && <span className="text-sm text-slate-500">Aucune section.</span>}
                            {sections.map((s) => (
                              <span key={s.id} className="rounded-full bg-white px-3 py-1.5 text-sm font-medium text-slate-700 ring-1 ring-slate-200">
                                {s.name}
                              </span>
                            ))}
                          </div>
                          <div className="mt-4 flex max-w-md gap-3">
                            <input
                              value={newSection}
                              onChange={(e) => setNewSection(e.target.value)}
                              placeholder="Nom de la section (ex. Groupe B)"
                              className={`${inputCls} mt-0`}
                            />
                            <button onClick={() => addSection(c.id)} disabled={saving} className={primaryBtn}>
                              Ajouter
                            </button>
                          </div>
                        </>
                      )}
                    </td>
                  </tr>
                )}
              </Fragment>
            ))}
          </tbody>
        </table>
      </div>
    </AdminShell>
  );
}

  subjects: number;
  courses: number;
  documents: number;
  classes: number;
  students: number;
  unreadNotifications: number;
  upcomingSessions: Session[];
};

  const [error, setError] = useState<string | null>(null);

    apiFetch<Dashboard>("/api/professor/dashboard")

        if (e instanceof ApiError && e.status !== 401) setError(e.message);

            <StatCard label="Matières" value={data.subjects} tone="green" />

"use client";

import { useEffect, useState } from "react";
import { ProfessorShell } from "@/components/ProfessorShell";
import { ErrorBox, PageHeader, tableWrap, theadCls } from "@/components/ui";
import { ApiError, apiFetch } from "@/lib/api";

type ClassItem = {
  id: number;
  name: string;
  level: string | null;
  academicYear: string;
  subjects: { id: number; name: string; code: string }[];
  students: { id: number; name: string; studentNumber: string; section: string | null }[];
};

export default function ProfessorClassesPage() {
  const [classes, setClasses] = useState<ClassItem[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    apiFetch<ClassItem[]>("/api/professor/classes")
      .then(setClasses)
      .catch((e) => {
        if (e instanceof ApiError && e.status !== 401) setError(e.message);
      })
      .finally(() => setLoading(false));
  }, []);

  return (
    <ProfessorShell>
      <PageHeader title="Mes classes" subtitle="Les classes qui suivent vos matières, avec leurs étudiants inscrits." />
      <ErrorBox message={error} />
      {loading && <p className="mt-6 text-slate-500">Chargement...</p>}
      {!loading && classes.length === 0 && (
        <p className="mt-6 rounded-2xl border border-slate-200 bg-white p-6 text-sm text-slate-500">
          Aucune classe pour le moment : elles apparaissent quand une de vos matières est rattachée à une classe.
        </p>
      )}
      <div className="mt-6 space-y-6">
        {classes.map((c) => (
          <section key={c.id} className="rounded-2xl border border-slate-200 bg-white p-6 shadow-sm">
            <div className="flex flex-wrap items-center justify-between gap-3">
              <div>
                <h2 className="text-lg font-semibold">{c.name}</h2>
                <p className="text-sm text-slate-500">{c.level ?? "-"} · {c.academicYear}</p>
              </div>
              <div className="flex flex-wrap gap-2">
                {c.subjects.map((s) => (
                  <span key={s.id} className="rounded-md bg-emerald-100 px-2 py-1 text-xs font-bold text-emerald-700">
                    {s.code} · {s.name}
                  </span>
                ))}
              </div>
            </div>
            <div className={tableWrap}>
              <table className="w-full text-left text-sm">
                <thead className={theadCls}>
                  <tr>
                    <th className="px-5 py-3">Étudiant</th>
                    <th className="px-5 py-3">N° étudiant</th>
                    <th className="px-5 py-3">Section</th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-100">
                  {c.students.length === 0 && (
                    <tr><td colSpan={3} className="px-5 py-6 text-center text-slate-500">Aucun étudiant inscrit.</td></tr>
                  )}
                  {c.students.map((s) => (
                    <tr key={s.id}>
                      <td className="px-5 py-3 font-medium text-slate-900">{s.name}</td>
                      <td className="px-5 py-3 text-slate-600">{s.studentNumber}</td>
                      <td className="px-5 py-3 text-slate-600">{s.section ?? "-"}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </section>
        ))}
      </div>
    </ProfessorShell>
  );
}

"use client";

import Link from "next/link";
import { useEffect, useState } from "react";
import { StudentShell } from "@/components/StudentShell";
import { useMe } from "@/components/RoleShell";
import { Panel, StatCard } from "@/components/Stats";
import { UpcomingSessions } from "@/components/UpcomingSessions";
import { ErrorBox, PageHeader, smallBtn } from "@/components/ui";
import { ApiError, apiFetch } from "@/lib/api";
import type { Session } from "@/lib/types";

type Dashboard = {
  subjects: number;
  overallAverage: number | null;
  absences: number;
  lates: number;
  unreadNotifications: number;
  upcomingSessions: Session[];
  latestCourses: { id: number; title: string; subjectName: string; createdAt: string }[];
};

function Content() {
  const me = useMe();
  const [data, setData] = useState<Dashboard | null>(null);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    apiFetch<Dashboard>("/api/student/dashboard")
      .then(setData)
      .catch((e) => {
        if (e instanceof ApiError && e.status !== 401) setError(e.message);
      });
  }, []);

  return (
    <>
      <PageHeader title={`Bonjour ${me.firstName}`} subtitle="Votre vie universitaire en un coup d'œil." />
      <ErrorBox message={error} />
      {!data && !error && <p className="mt-6 text-slate-500">Chargement...</p>}
      {data && (
        <>
          <div className="mt-6 grid gap-4 sm:grid-cols-2 lg:grid-cols-5">
            <StatCard label="Matières" value={data.subjects} tone="blue" />
            <StatCard
              label="Moyenne générale"
              value={data.overallAverage == null ? "-" : `${data.overallAverage} / 20`}
              tone={data.overallAverage == null ? "slate" : data.overallAverage < 10 ? "red" : "green"}
            />
            <StatCard label="Absences" value={data.absences} tone={data.absences > 0 ? "red" : "slate"} />
            <StatCard label="Retards" value={data.lates} tone={data.lates > 0 ? "orange" : "slate"} />
            <Link href="/student/notifications" className="contents">
              <StatCard
                label="Notifications non lues"
                value={data.unreadNotifications}
                tone={data.unreadNotifications > 0 ? "orange" : "slate"}
              />
            </Link>
          </div>
          <div className="mt-6 grid gap-6 xl:grid-cols-2">
            <Panel title="Prochaines séances">
              <UpcomingSessions sessions={data.upcomingSessions} showProfessor />
              <Link href="/student/timetable" className={`mt-4 inline-block ${smallBtn}`}>Voir tout l&apos;emploi du temps</Link>
            </Panel>
            <Panel title="Derniers cours">
              {data.latestCourses.length === 0 ? (
                <p className="text-sm text-slate-500">Aucun cours disponible pour le moment.</p>
              ) : (
                <ul className="space-y-3">
                  {data.latestCourses.map((c) => (
                    <li key={c.id} className="rounded-xl bg-slate-50 px-4 py-3 text-sm">
                      <p className="font-medium text-slate-900">{c.title}</p>
                      <p className="text-xs text-slate-500">
                        {c.subjectName} · {new Date(c.createdAt).toLocaleDateString("fr-FR", { day: "2-digit", month: "short" })}
                      </p>
                    </li>
                  ))}
                </ul>
              )}
              <Link href="/student/courses" className={`mt-4 inline-block ${smallBtn}`}>Tous mes cours</Link>
            </Panel>
          </div>
        </>
      )}
    </>
  );
}

export default function StudentHome() {
  return (
    <StudentShell>
      <Content />
    </StudentShell>
  );
}
